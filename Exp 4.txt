class TextEditor:
    def __init__(self):
        self.undo_stack = []
        self.redo_stack = []
        self.current_state = ""

    def make_change(self, new_text):
        self.undo_stack.append(self.current_state)
        self.current_state = new_text
        self.redo_stack.clear()
        print("✅ Change made.")

    def undo(self):
        if self.undo_stack:
            self.redo_stack.append(self.current_state)
            self.current_state = self.undo_stack.pop()
            print("↩️ Undo performed.")
        else:
            print("⚠️ Nothing to undo.")

    def redo(self):
        if self.redo_stack:
            self.undo_stack.append(self.current_state)
            self.current_state = self.redo_stack.pop()
            print("🔁 Redo performed.")
        else:
            print("⚠️ Nothing to redo.")

    def display(self):
        print(f"📄 Current Document State: '{self.current_state}'")


# ----------------- MAIN PROGRAM -----------------
def main():
    editor = TextEditor()

    while True:
        print("\n--- TEXT EDITOR MENU ---")
        print("1. Make a Change")
        print("2. Undo")
        print("3. Redo")
        print("4. Display Document State")
        print("5. Exit")

        choice = input("Enter your choice (1-5): ")

        if choice == '1':
            new_text = input("Enter new text content: ")
            editor.make_change(new_text)
        elif choice == '2':
            editor.undo()
        elif choice == '3':
            editor.redo()
        elif choice == '4':
            editor.display()
        elif choice == '5':
            print("🚪 Exiting the text editor.")
            break
        else:
            print("❌ Invalid choice. Please enter a number from 1 to 5.")

if __name__ == "__main__":
    main()
