class Node:
    def __init__(self, key):
        # Store the key (value) of the node
        self.key = key
        # Initialize the left child as None
        self.left = None
        # Initialize the right child as None
        self.right = None

class BST:
    def __init__(self):
        # Initially, the tree has no root node
        self.root = None

    # Insert a new node into the BST
    def insert(self, root, key):
        # If tree is empty, create a new node
        if root is None:
            return Node(key)
        # If key is smaller than root, insert in left subtree
        if key < root.key:
            root.left = self.insert(root.left, key)
        # If key is larger than root, insert in right subtree
        elif key > root.key:
            root.right = self.insert(root.right, key)
        # Return unchanged root
        return root

    # Search for a key in the BST
    def search(self, root, key):
        # If tree is empty or key matches root key
        if root is None or root.key == key:
            return root
        # Search in left subtree if key is smaller
        if key < root.key:
            return self.search(root.left, key)
        # Otherwise, search in right subtree
        return self.search(root.right, key)

    # Find the node with the minimum key value (used in deletion)
    def find_min(self, root):
        while root.left:
            root = root.left
        return root

    # Delete a node from the BST
    def delete(self, root, key):
        if root is None:
            return root
        # Traverse the tree to find the node to delete
        if key < root.key:
            root.left = self.delete(root.left, key)
        elif key > root.key:
            root.right = self.delete(root.right, key)
        else:
            # Node found — handle deletion cases
            if root.left is None:
                return root.right
            elif root.right is None:
                return root.left
            # Node with two children
            temp = self.find_min(root.right)
            root.key = temp.key
            root.right = self.delete(root.right, temp.key)
        return root

    # Display the BST using inorder traversal
    def inorder(self, root):
        if root:
            self.inorder(root.left)
            print(root.key, end=" ")
            self.inorder(root.right)

# Create an instance of BST
bst = BST()
root = None

# Insert elements into the BST
for key in [50, 30, 70, 20, 40, 60, 80]:
    root = bst.insert(root, key)

print("Inorder traversal of the BST:")
bst.inorder(root)

# Search for a key
key_to_search = 60
result = bst.search(root, key_to_search)
if result:
    print(f"\nKey {key_to_search} found in BST.")
else:
    print(f"\nKey {key_to_search} not found in BST.")

# Delete a node
key_to_delete = 30
root = bst.delete(root, key_to_delete)
print(f"\nInorder traversal after deleting {key_to_delete}:")
bst.inorder(root)
